import { sha256 } from "@noble/hashes/sha2.js";
import { concatBytes, u32be, utf8 } from "./bytes.ts";

// Domain-separated, length-framed hash (SIGE spec section 6.1, demo profile).
export function dhash(domain: string, ...parts: Uint8Array[]): Uint8Array {
  const framed = [utf8("SIGE-DEMO/v1"), u32be(domain.length), utf8(domain)];

}

```ts
import { bls12_381 } from "@noble/curves/bls12-381.js";
import { bytesToBigInt, concatBytes, randomBytes, u32be, utf8 } from "./bytes.ts";
import { kdf } from "./hash.ts";

// Two-gate Boneh-Franklin-style IBE KEM (SIGE spec sections 5.2-5.4).
// Demo profile: CPA-level composition, as recorded in spec section 19.

const G2 = bls12_381.G2.Point;
const Fr = bls12_381.fields.Fr;
const Fp12 = bls12_381.fields.Fp12;

export type Gate = "warrant" | "log";

const DST: Record<Gate, string> = {
  warrant: "SIGE-DEMO-WARRANT-V1",
  log: "SIGE-DEMO-LOG-V1",
};

export type G2Point = typeof G2.BASE;

export interface EpochKeys {
  epoch: number;
  xA: bigint; // warrant master secret, HSM domain A
  xB: bigint; // publication master secret, HSM domain B
  pkA: G2Point;
  pkB: G2Point;
}

export function randomScalar(): bigint {
  const s = Fr.create(bytesToBigInt(randomBytes(48)));
  return s === 0n ? 1n : s;
}

export function genEpoch(epoch: number): EpochKeys {
  const xA = randomScalar();
  const xB = randomScalar();
  return { epoch, xA, xB, pkA: G2.BASE.multiply(xA), pkB: G2.BASE.multiply(xB) };
}

// Account-scoped identity string (spec section 5.3). Fixed at enrollment.
export function gateIdentity(
  gate: Gate,
  networkId: Uint8Array,
  accountId: Uint8Array,
  enrollmentId: Uint8Array,
  epoch: number,
): Uint8Array {
  const label = gate === "warrant" ? "SIGE/v1/warrant" : "SIGE/v1/log";
  return concatBytes(
    utf8(label),
    u32be(networkId.length),
    networkId,
    u32be(accountId.length),
    accountId,
    u32be(enrollmentId.length),
    enrollmentId,
    u32be(epoch),
  );
}

function hashToG1(id: Uint8Array, gate: Gate) {
  return bls12_381.G1.hashToCurve(id, { DST: DST[gate] });
}

function gtBytes(z: ReturnType<typeof bls12_381.pairing>): Uint8Array {
  return Fp12.toBytes(z);
}

export interface Encapsulation {
  U: Uint8Array; // compressed G2 point
  K: Uint8Array; // 32-byte AEAD key
}

// Client side. Concatenated pairing results, never multiplied (spec section 5.4).
export function encapsulate(
  ids: { warrant: Uint8Array; log: Uint8Array },
  keys: { pkA: G2Point; pkB: G2Point },
  transcriptHash: Uint8Array,
  context: Uint8Array,
): Encapsulation {
  const r = randomScalar();
  const U = G2.BASE.multiply(r);
  const zA = bls12_381.pairing(hashToG1(ids.warrant, "warrant").multiply(r), keys.pkA);
  const zB = bls12_381.pairing(hashToG1(ids.log, "log").multiply(r), keys.pkB);
  const K = kdf(concatBytes(gtBytes(zA), gtBytes(zB)), transcriptHash, context);
  return { U: U.toBytes(true), K };
}

// HSM side. Returns a pairing contribution, never a reusable private key (spec section 5.6).
export function deriveContribution(
  gate: Gate,
  id: Uint8Array,
  masterSecret: bigint,
  U: Uint8Array,
): Uint8Array {
  const d = hashToG1(id, gate).multiply(masterSecret);
  return gtBytes(bls12_381.pairing(d, G2.fromBytes(U)));
}

// Ceremony side. K exists only when both contributions are present.
export function combineContributions(
  zA: Uint8Array,
  zB: Uint8Array,
  transcriptHash: Uint8Array,
  context: Uint8Array,
): Uint8Array {
  return kdf(concatBytes(zA, zB), transcriptHash, context);
}
```

```ts
// The reframe's centerpiece. The external authority must publish its records
// once a process closes. A keyless verifier joins every public unseal leaf
// against that docket. Past the horizon, an unmatched leaf is flagged,
// never auto-revealed (SIGE spec section 8.2: flags, not reveals).
export type ReconcileStatus = "matched" | "pending" | "overdue";

export interface DocketRecord {
  authorizationHash: string;
  publishedAtHeight: number;
  caseSummary: string;
}

export interface UnsealSighting {
  authorizationHash: string;
  anchorHeight: number;
}

export interface ReconcileEntry {
  authorizationHash: string;
  status: ReconcileStatus;
}

export interface DocketHorizon {
  tipHeight: number;
  horizonBlocks: number;
}

export function reconcile(
  sightings: UnsealSighting[],
  docket: DocketRecord[],
  horizon: DocketHorizon,
): ReconcileEntry[] {
  const published = new Set(docket.map((r) => r.authorizationHash));
  return sightings.map((s) => {
    if (published.has(s.authorizationHash)) {
      return { authorizationHash: s.authorizationHash, status: "matched" };
    }
    const age = horizon.tipHeight - s.anchorHeight;
    return {
      authorizationHash: s.authorizationHash,
      status: age > horizon.horizonBlocks ? "overdue" : "pending",
    };
  });
}
```

import { buildScenarios } from "../scenarios/scenarios.ts";
import { GENERIC, PROFILES } from "../world/profile.ts";

interface RunMessage {
  type: "run";
  id: string;
  profileId: string;
}

self.onmessage = async (event: MessageEvent<RunMessage>) => {
  const { id, profileId } = event.data;
  const profile = PROFILES[profileId] ?? GENERIC;
  const scenario = buildScenarios(profile).find((s) => s.id === id);
  if (!scenario) {
    self.postMessage({ type: "error", message: `unknown scenario: ${id}` });
    return;
  }
  try {
    const ok = await scenario.run(
      (step) => self.postMessage({ type: "step", step }),
      (done, total) => self.postMessage({ type: "progress", done, total }),
    );
    self.postMessage({ type: "done", ok });
  } catch (err) {
    self.postMessage({ type: "error", message: String(err) });
  }
};

  root: HTMLElement,

  hooks: RunnerHooks,

      <a href="#/" class="back">← ledger</a>

  const steps = root.querySelector<HTMLOListElement>("#steps")!;
  const verdict = root.querySelector<HTMLParagraphElement>("#verdict")!;
  const progress = root.querySelector<HTMLProgressElement>("#delay")!;
  const button = root.querySelector<HTMLButtonElement>("#run")!;
  let worker: Worker | null = null;

  const appendStep = (step: DemoStep) => {
    const li = document.createElement("li");
    li.className = step.ok ? "ok" : "fail";
    li.textContent = `${step.ok ? "✓" : "✗"} ${step.label}${step.detail ? ` (${step.detail})` : ""}`;
    steps.append(li);
  };

  button.addEventListener("click", () => {
    worker?.terminate();
    steps.innerHTML = "";
    verdict.textContent = "";
    progress.hidden = true;
    button.disabled = true;
    worker = new Worker(new URL("./worker.ts", import.meta.url), { type: "module" });
    worker.onmessage = (event) => {
      const msg = event.data;
      if (msg.type === "step") appendStep(msg.step);
      if (msg.type === "progress") {
        progress.hidden = false;
        progress.value = msg.done / msg.total;
      }
      if (msg.type === "done") {
        progress.hidden = true;
        verdict.textContent = msg.ok ? "scenario passed" : "SCENARIO FAILED";
        verdict.className = msg.ok ? "verdict ok" : "verdict fail";
        button.disabled = false;
      }
      if (msg.type === "error") {
        verdict.textContent = `error: ${msg.message}`;
        verdict.className = "verdict fail";
        button.disabled = false;
      }
    };
    worker.postMessage({ type: "run", id: scenario.id, profileId });
  });

  root.querySelector<HTMLAnchorElement>(".back")!.addEventListener("click", hooks.onBack);
}
